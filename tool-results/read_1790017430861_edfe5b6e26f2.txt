   320→   320→function shade(hex, amt) {
   321→   321→  if (!isHex(hex)) return '#666';
   322→   322→  var r = parseInt(hex.substr(1,2),16), g = parseInt(hex.substr(3,2),16), b = parseInt(hex.substr(5,2),16);
   323→   323→  return 'rgb(' + Math.round(r*amt) + ',' + Math.round(g*amt) + ',' + Math.round(b*amt) + ')';
   324→   324→}
   325→   325→// Franjas del trapo con el segundo color del club. Si los dos colores son el
   326→   326→// mismo (o falta el segundo), se usa un tono mas oscuro para que la bandera no

   328→   328→function franjas(c, alto) {

   330→   330→  var b = (isHex(c.color2) && c.color2.toLowerCase() !== a.toLowerCase()) ? c.color2 : shade(a, 0.62);
