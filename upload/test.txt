<script>
// El aforo lo manda el servidor (/api/config). Este valor es solo el respaldo
// por si la llamada falla: la fuente de verdad esta en server.js.
var CAPACITY = 100000;

function nf(n) { return (n || 0).toLocaleString('es-AR'); }
function pf(n) { return String(n).replace('.', ',') + '%'; }
function hinchas(n) { return nf(n) + (n === 1 ? ' hincha' : ' hinchas'); }
function esc(s) {
  return String(s).replace(/[&<>"']/g, function(c) {
    return { '&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;' }[c];
  });
}
function isHex(h) { return typeof h === 'string' && /^#[0-9a-fA-F]{6}$/.test(h); }
function colorDe(c) { return isHex(c.color) ? c.color : '#8a8a8a'; }
// Para cuadraditos y porcentajes sobre el panel crema. El estadio y las
// banderas siguen usando colorDe(), que es el color real del club.
function colorLegible(c) { return window.ColorClub(c.color, c.color2); }
// Mismo desempate que el ranking completo, para que las dos pantallas coincidan.
function ordenRanking(a, b) {
  if (b.count !== a.count) return b.count - a.count;
  return a.name.localeCompare(b.name, 'es');
}
// Saca la barra gris de carga y escribe el valor de verdad.
function llenar(id, texto) {
  var el = document.getElementById(id);
  el.classList.remove('cargando');
  el.textContent = texto;
}
function shade(hex, amt) {
  if (!isHex(hex)) return '#666';
  var r = parseInt(hex.substr(1,2),16), g = parseInt(hex.substr(3,2),16), b = parseInt(hex.substr(5,2),16);
  return 'rgb(' + Math.round(r*amt) + ',' + Math.round(g*amt) + ',' + Math.round(b*amt) + ')';
}
// Franjas del trapo con el segundo color del club. Si los dos colores son el
// mismo (o falta el segundo), se usa un tono mas oscuro para que la bandera no
// quede como un rectangulo liso.
function franjas(c, alto) {
  var a = colorDe(c);
  var b = (isHex(c.color2) && c.color2.toLowerCase() !== a.toLowerCase()) ? c.color2 : shade(a, 0.62);
  return 'repeating-linear-gradient(180deg, transparent 0 ' + alto + 'px, ' + b + ' ' + alto + 'px ' + (alto*2) + 'px)';
}
function timeAgo(iso) {
  var m = Math.round((Date.now() - new Date(iso).getTime()) / 60000);
  if (m < 1) return 'recién';
  if (m < 60) return 'hace ' + m + ' min';
  var h = Math.round(m / 60);
  if (h < 24) return 'hace ' + h + ' h';
  return 'hace ' + Math.round(h / 24) + ' d';
}


Promise.all([
  fetch('/api/clubes').then(function(r) { return r.json(); }),
  fetch('/api/feed').then(function(r) { return r.json(); }),
  fetch('/api/config').then(function(r) { return r.json(); }).catch(function() { return null; })
]).then(function(res) {
  var raw = res[0], feed = res[1], cfg = res[2];
  if (cfg && cfg.capacidad) CAPACITY = cfg.capacidad;
  var vendidas = raw.reduce(function(s, c) { return s + (c.hinchas || 0); }, 0);
  var libres = Math.max(0, CAPACITY - vendidas);
  var pct = Math.round(vendidas / CAPACITY * 10000) / 100;

  var clubs = raw.map(function(c) {
    return {
      name: c.club, apodo: c.apodo, file: c.archivo_escudo,
      color: c.color_principal, color2: c.color_secundario,
      count: c.hinchas || 0,
      value: Math.round((c.hinchas || 0) / CAPACITY * 10000) / 100
    };
  });
  var ranked = clubs.slice().sort(ordenRanking);
  ranked.forEach(function(c, i) { c.rank = i + 1; });
  var conHinchas = ranked.filter(function(c) { return c.count > 0; });

  // ----- cifras -----
  llenar('totalNum', nf(vendidas));
  llenar('faltanNum', nf(libres));
  llenar('aforoPct', pf(pct) + ' llena');
  document.getElementById('aforoFill').style.width = Math.max(vendidas > 0 ? 0.6 : 0, pct) + '%';

  // ----- cancha llena: se acabo la edicion -----
  // Cuando no quedan lugares no se puede seguir invitando a comprar: el boton
  // llevaria a un formulario que va a fallar. Se cambia por el resultado final.
  if (libres <= 0) {
    document.querySelector('.faltan').innerHTML = '<b>La cancha se llenó.</b> Este es el resultado final.';
    var cta = document.querySelector('.cta-card .card-body');
    if (cta) {
      cta.innerHTML =
        '<div class="cta-t1 pix">Se llenó</div>' +
        '<div class="cta-t2 pix">edición cerrada</div>' +
        '<a class="cta-btn pix" href="estadisticas.html">Ver resultado final</a>';
    }
    document.getElementById('memeBanner').style.display = 'none';
  }

  // ----- top 3 + banderas -----
  var top3 = document.getElementById('top3');
  var flags = document.getElementById('flags');
  var etiquetas = ['1º', '2º', '3º'];
  if (!conHinchas.length) {
    top3.innerHTML = '<div class="empty-note">Todavía no entró nadie al estadio.</div>';
  }
  conHinchas.slice(0, 3).forEach(function(c, i) {
    var row = document.createElement('div');
    row.className = 'top-row';
    row.style.cursor = 'pointer';
    row.innerHTML =
      '<div class="top-pos pix">' + etiquetas[i] + '</div>' +
      '<div class="sw" style="background:' + colorLegible(c) + '"></div>' +
      '<div class="top-info"><div class="top-name">' + esc(c.apodo || c.name) + '</div></div>' +
      '<div class="top-pct pix" style="color:' + colorLegible(c) + '">' + (c.value > 0 ? pf(c.value) : '&lt;0,01%') + '</div>';
    row.addEventListener('click', function() { verClub(c); });
    top3.appendChild(row);

    var f = document.createElement('div');
    f.className = 'flag f' + (i + 1);
    var alto = [15, 12, 10][i];
    f.innerHTML =
      '<div class="cloth" style="background:' + colorDe(c) + '">' +
        '<div class="bands" style="background:' + franjas(c, alto) + '"></div>' +
        '<div class="cloth-top">' +
          '<div class="num pix">' + etiquetas[i] + '</div>' +
          '<img class="crest" src="escudos/' + esc(c.file) + '" alt="">' +
        '</div>' +
        '<div class="nm pix">' + esc(c.apodo || c.name) + '</div>' +
      '</div><div class="pole"></div>';
    f.title = c.name;
    f.addEventListener('click', function() { verClub(c); });
    flags.appendChild(f);
  });

  // ----- banner del hincha: solo aparece si el archivo meme.png existe -----
  var memeImg = document.getElementById('memeImg');
  function mostrarMeme() {
    // Con la cancha llena no se invita a sumar: la imagen puede cargar despues
    // de que se aplico el estado de cierre y volveria a mostrar el banner.
    if (libres <= 0) return;
    var lider = conHinchas[0];
    document.getElementById('memeBanner').style.display = 'flex';
    if (lider) {
      document.getElementById('memeT1').textContent = '¿Vas a dejar que te pasen?';
      document.getElementById('memeT2').textContent =
        (lider.apodo || lider.name) + ' va primero con ' + hinchas(lider.count) + '. ' +
        'Quedan ' + nf(libres) + ' lugares: cada uno que no llena tu club, se lo queda otro.';
    } else {
      document.getElementById('memeT1').textContent = 'La cancha está vacía';
      document.getElementById('memeT2').textContent =
        'Las ' + nf(CAPACITY) + ' hinchas están libres. El primero que entre, manda.';
    }
  };
  function ocultarMeme() { document.getElementById('memeBanner').style.display = 'none'; }
  // Si la imagen ya venia cacheada, el evento load ya paso y nunca se dispara:
  // por eso se chequea complete/naturalWidth antes de quedarse esperando.
  if (memeImg.complete) {
    memeImg.naturalWidth > 0 ? mostrarMeme() : ocultarMeme();
  } else {
    memeImg.onload = mostrarMeme;
    memeImg.onerror = ocultarMeme;
  }

  // ----- ranking lateral: todos los clubes que ya tienen hinchas -----
  var rankList = document.getElementById('rankList');
  if (!conHinchas.length) {
    rankList.innerHTML = '<div class="empty-note">Todavía no entró nadie a la cancha.</div>';
  }
  conHinchas.forEach(function(c) {
    var d = document.createElement('div');
    d.className = 'rank-row' + (c.rank <= 3 ? ' podio' : '');
    d.innerHTML =
      '<div class="rank-pos pix">' + c.rank + 'º</div>' +
      '<div class="sw" style="background:' + colorLegible(c) + '"></div>' +
      '<div class="rank-nm">' + esc(c.apodo || c.name) + '</div>' +
      '<div class="rank-pct pix" style="color:' + colorLegible(c) + '">' + (c.value > 0 ? pf(c.value) : '&lt;0,01%') + '</div>';
    d.addEventListener('click', function() { verClub(c); });
    rankList.appendChild(d);
  });

  // ----- estadio -----
  var canvas = document.getElementById('estadioCanvas');
  var nota = document.getElementById('stageNote');
  var view;

  function conectar() {
    canvas.addEventListener('click', function(ev) {
      var r = canvas.getBoundingClientRect(), k = view.width / r.width;
      var c = view.clubAt((ev.clientX - r.left) * k, (ev.clientY - r.top) * k);
      if (c) verClub(c);
    });
    canvas.addEventListener('mousemove', function(ev) {
      var r = canvas.getBoundingClientRect(), k = view.width / r.width;
      var c = view.clubAt((ev.clientX - r.left) * k, (ev.clientY - r.top) * k);
      canvas.title = c ? c.name + (c.apodo ? ' — ' + c.apodo : '') + '\n' + hinchas(c.count) : '';
    });
  }

  view = Estadio.render(canvas, { clubs: ranked, capacity: CAPACITY });
  nota.textContent = 'vista general de la cancha · los colores se actualizan en vivo';
  conectar();

  // ----- ficha de club -----
  function verClub(c) {
    var el = document.getElementById('clubDetail');
    var arriba = ranked.filter(function(o) { return o.count > c.count; }).pop();
    var brecha;
    if (arriba) {
      brecha = esc(c.name) + ' necesita <b>' + hinchas(arriba.count - c.count + 1) + '</b> para pasar a ' + esc(arriba.name);
    } else if (c.count > 0) {
      brecha = 'Es la hinchada que más cancha llena. Defendela.';
    } else {
      brecha = 'Todavía no metió ni un hincha. <b>Poné el primero.</b>';
    }
    el.innerHTML =
      '<img class="club-crest" src="escudos/' + c.file + '" alt="">' +
      '<div class="club-name pix">' + esc(c.name) + '</div>' +
      '<div class="club-apodo">' + esc(c.apodo || '') + '</div>' +
      '<div class="club-stat"><span>Puesto</span><span>' + (c.count > 0 ? c.rank + 'º' : '—') + '</span></div>' +
      '<div class="club-stat"><span>De la cancha</span><span>' + pf(c.value) + '</span></div>' +
      '<div class="club-stat"><span>Hinchas</span><span>' + nf(c.count) + '</span></div>' +
      '<div class="gap-note">' + brecha + '</div>' +
      '<a class="cta-btn pix" href="comprar.html?club=' + encodeURIComponent(c.file) + '">Sumarme a ' + esc(c.name) + '</a>' +
      '<a class="card-link" href="#" id="cerrarClub">« Volver al ranking</a>';
    document.getElementById('cardComo').style.display = 'none';
    document.getElementById('cardRef').style.display = 'none';
    document.getElementById('cardClub').style.display = 'block';
    document.getElementById('cerrarClub').addEventListener('click', function(e) {
      e.preventDefault();
      document.getElementById('cardClub').style.display = 'none';
      document.getElementById('cardComo').style.display = 'block';
      document.getElementById('cardRef').style.display = 'block';
    });
  }

  // ----- llegada desde el pago: cartel de "listo" y ficha del club -----
  // Se muestra solo si la orden esta pagada de verdad: el cartel no sale de la URL.
  var listo = new URLSearchParams(location.search).get('listo');
  if (listo) {
    fetch('/api/orden?id=' + encodeURIComponent(listo)).then(function(r) { return r.json(); }).then(function(o) {
      if (!o.ok || o.estado !== 'pagada') return;
      var n = o.asignadas != null ? o.asignadas : o.cantidad;
      var b = document.getElementById('listoBanner');
      b.innerHTML = '<span class="t">✔</span><span></span>';
      b.lastChild.textContent = (n === 1 ? '¡Listo! Tu hincha ya está en la cancha' : '¡Listo! Tus ' + nf(n) + ' hinchas ya están en la cancha') +
        (o.club ? ' para ' + o.club : '');
      b.style.display = 'flex';
      var c = ranked.filter(function(x) { return x.file === o.archivo_escudo; })[0];
      if (c) verClub(c);
    }).catch(function() {});
  }

  // ----- llegada con el link de un club (/c/talleres): cartel con su puesto -----
  var deClub = new URLSearchParams(location.search).get('club');
  var cc = deClub && !listo ? ranked.filter(function(x) { return x.file === deClub; })[0] : null;
  if (cc) {
    var arribaDe = ranked.filter(function(o) { return o.count > cc.count; }).pop();
    var situacion;
    if (!cc.count) situacion = 'Todavía no metieron ni un hincha. <b>¡Poné el primero!</b>';
    else if (!arribaDe) situacion = 'Van <b>primeros</b> con ' + hinchas(cc.count) + '. Que no los pasen.';
    else situacion = 'Van <b>' + cc.rank + 'º</b> con ' + hinchas(cc.count) + '. A <b>' + hinchas(arribaDe.count - cc.count + 1) +
                     '</b> de pasar a ' + esc(arribaDe.name) + '.';
    var bc = document.getElementById('listoBanner');
    bc.className = 'listo-banner club pix';
    bc.innerHTML =
      '<img src="escudos/' + encodeURIComponent(cc.file) + '" alt="">' +
      '<div class="txt">Hinchas de ' + esc(cc.name) + ': ' + situacion + '</div>' +
      '<a class="cta-btn red pix" href="comprar.html?club=' + encodeURIComponent(cc.file) + '">Sumar a ' + esc(cc.name) + '</a>';
    bc.style.display = 'flex';
    verClub(cc);
    // El boton fijo del celular tambien apunta a ese club.
    var fija = document.querySelector('.cta-fija');
    if (fija) { fija.href = 'comprar.html?club=' + encodeURIComponent(cc.file); fija.textContent = '¡Meto hinchas para ' + cc.name + '!'; }
  }

  // ----- feed -----
  var fl = document.getElementById('feedList');
  if (!feed.length) {
    fl.innerHTML = '<div class="empty-note">Acá van a aparecer las últimas compras.</div>';
  } else {
    feed.slice(0, 8).forEach(function(f) {
      var row = document.createElement('div');
      row.className = 'feed-row';
      row.innerHTML =
        '<div class="feed-av"><img src="escudos/' + f.archivo_escudo + '" alt=""></div>' +
        '<div style="flex:1;min-width:0">' +
          '<div class="feed-t"></div>' +
          (f.mensaje ? '<div class="feed-m">"' + esc(f.mensaje) + '"</div>' : '') +
        '</div>' +
        '<div class="feed-time">' + timeAgo(f.fecha) + '</div>';
      row.querySelector('.feed-t').textContent = f.nombre + ' → x' + f.cantidad + ' ' + f.club;
      fl.appendChild(row);
    });
  }
}).catch(function(err) {
  // Error recuperable: se conserva la estructura de la pagina y se ofrece
  // reintentar, en vez de dejar la cancha vacia sin explicacion.
  console.error('no se pudieron cargar los datos:', err);
  document.querySelector('.grid').insertAdjacentHTML('beforebegin',
    '<div class="card error-datos"><div class="card-head pix">No pudimos actualizar la cancha</div>' +
    '<div class="card-body">' +
    '<p>Puede ser tu conexión o algo de nuestro lado. Los datos que ves pueden estar viejos.</p>' +
    '<button type="button" class="cta-btn pix" id="reintentar">Reintentar</button>' +
    '</div></div>');
  document.getElementById('reintentar').addEventListener('click', function() {
    this.textContent = 'Reintentando...';
    this.disabled = true;
    location.reload();
  });
});
</script>
